if (isset($_GET["ris"]))
{
    // ================= FASE 2: CORREZIONE =================
    $a   = $_GET["a"];       // primi numeri    (dagli hidden)
    $b   = $_GET["b"];       // secondi numeri  (dagli hidden)
    $op  = $_GET["oper"];    // operazioni      (dagli hidden)
    $ris = $_GET["ris"];     // risposte dell'utente
    $num = count($a);
    $giuste = 0;

    for ($i = 0; $i < $num; $i++)
    {
        $x = $a[$i];
        $y = $b[$i];
        settype($x, "integer");
        settype($y, "integer");

        // ricalcolo il risultato giusto (NON lo avevo messo nell'hidden)
        if ($op[$i] == "somma")            { $giusto = $x + $y; $segno = "+"; }
        elseif ($op[$i] == "sottrazione")  { $giusto = $x - $y; $segno = "-"; }
        else                               { $giusto = $x * $y; $segno = "x"; }

        // 6) facoltativo: preparo il contatore degli errori per operazione
        if (!isset($errori[$op[$i]])) $errori[$op[$i]] = 0;

        printf("%2d %s %2d = %3d", $x, $segno, $y, $giusto);
        if ($ris[$i] != "" && $ris[$i] == $giusto)
        {
            echo "   (hai risposto ".$ris[$i].")\n";
            $giuste++;
        }
        else
        {
            echo "   <font color=red>(hai risposto ".$ris[$i].")</font>\n";
            $errori[$op[$i]] = $errori[$op[$i]] + 1;
        }
    }

    // 5) punteggio e salvataggio
    $punteggio = round($giuste / $num * 100);
    echo "\nRisposte giuste: $giuste su $num\n";
    echo "Punteggio: $punteggio/100\n";
    file_put_contents("ultimo.txt", "$num,$giuste,$punteggio\n");
    echo "(risultato salvato in ultimo.txt)\n";

    // 6) errori per operazione
    echo "\nErrori per operazione:\n";
    foreach ($errori as $operazione => $n)
        echo "$operazione: $n\n";

    echo "\n<a href='index.html'>Nuovo allenamento</a>";
}
else

    for ($i = 0; $i < $num; $i++)

        $operazione = $operazioni[array_rand($operazioni)];   // operazione a caso
        $x = rand(1, $max);

        // nella sottrazione metto il numero piu' grande per primo (niente negativi)
        if ($operazione == "sottrazione" && $y > $x)
        {
            $tmp = $x;   $x = $y;   $y = $tmp;      // scambio
        }

        if ($operazione == "somma") $segno = "+";
        elseif ($operazione == "sottrazione") $segno = "-";
        else $segno = "x";

        printf("%2d %s %2d = ", $x, $segno, $y);
        echo "<input type='number' name='ris[]'>\n";
