<?php
// =========================================================
//  VERIFICA 1 - VOTI     (soluzione con PHP base)
//  Avvio: nella cartella "soluzione" -> php -S localhost:8000
// =========================================================

$righe = file("../voti.txt");   // il file dati e' nella cartella sopra

// ---------- LETTURA DEL FILE (Schema A) ----------
for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $studente = $parti[0];
    $materia  = $parti[1];
    $voto     = $parti[2];
    settype($voto, "integer");        // "7\n" -> 7

    // salvo la riga intera, mi serve per il punto 4 (Schema H)
    $tuttiVoti[] = array($studente, $materia, $voto);

    // per la media dello studente servono SOMMA e NUMERO di voti (Schema D)
    if (!isset($somma[$studente]))
    {
        $somma[$studente] = 0;
        $num[$studente] = 0;
    }
    $somma[$studente] = $somma[$studente] + $voto;
    $num[$studente]   = $num[$studente] + 1;

    // insufficienze per materia (Schema B: contare)
    if (!isset($insuff[$materia])) $insuff[$materia] = 0;
    if ($voto < 6) $insuff[$materia] = $insuff[$materia] + 1;

    // facoltativo: media per materia
    if (!isset($sommaMat[$materia]))
    {
        $sommaMat[$materia] = 0;
        $numMat[$materia] = 0;
    }
    $sommaMat[$materia] = $sommaMat[$materia] + $voto;
    $numMat[$materia]   = $numMat[$materia] + 1;
}

// calcolo le medie in un nuovo array associativo  studente => media
foreach ($somma as $studente => $s)
    $media[$studente] = $s / $num[$studente];

// ---------- 1) MEDIA DI OGNI STUDENTE ----------
echo "<h2>1) Media di ogni studente</h2>";
echo "<table border=1><tr><th>Studente</th><th>Media</th></tr>";
foreach ($media as $studente => $m)
    printf("<tr><td>%s</td><td>%.2f</td></tr>", $studente, $m);
echo "</table>";

// ---------- 2) MEDIA PIU' ALTA (Schema E, con pari merito) ----------
$max = -1;
foreach ($media as $studente => $m)
    if ($m > $max) $max = $m;

echo "<h2>2) Media piu' alta</h2>";
foreach ($media as $studente => $m)
    if ($m == $max) printf("%s (%.2f)<br>", $studente, $m);

// ---------- 3) INSUFFICIENZE PER MATERIA ----------
echo "<h2>3) Insufficienze per materia</h2>";
foreach ($insuff as $materia => $n)
    echo "$materia: $n<br>";

// ---------- 4) FORM: VOTI DI UNO STUDENTE (Schema I + H) ----------
echo "<h2>4) Voti di uno studente</h2>";
echo "<form action='' method='get'>";
echo "<select name='studente'>";
foreach ($media as $studente => $m)
    echo "<option value='$studente'>$studente</option>";
echo "</select> <input type='submit' value='Mostra voti'>";
echo "</form>";

if (isset($_GET["studente"]))
{
    $scelto = $_GET["studente"];
    echo "<p>Voti di <b>$scelto</b>:</p>";
    echo "<table border=1><tr><th>Materia</th><th>Voto</th></tr>";
    for ($i = 0; $i < count($tuttiVoti); $i++)
        if ($tuttiVoti[$i][0] == $scelto)
            echo "<tr><td>".$tuttiVoti[$i][1]."</td><td>".$tuttiVoti[$i][2]."</td></tr>";
    echo "</table>";
}

// ---------- 5) FILE pagella.html (Schema G + J) ----------
ksort($media);                       // ordine alfabetico per nome (la chiave)
$html = "<html><body><h1>Pagella</h1><table border=1>\n";
foreach ($media as $studente => $m)
{
    if ($m < 6)
        $html = $html . "<tr><td><font color=red>$studente</font></td><td><font color=red>" . round($m, 2) . "</font></td></tr>\n";
    else
        $html = $html . "<tr><td>$studente</td><td>" . round($m, 2) . "</td></tr>\n";
}
$html = $html . "</table></body></html>";
file_put_contents("pagella.html", $html);
echo "<h2>5) File creato</h2><a href='pagella.html'>Apri pagella.html</a>";

// ---------- 6) FACOLTATIVO: MATERIA CON LA MEDIA PIU' BASSA (minimo) ----------
$min = 999;
foreach ($sommaMat as $materia => $s)
{
    $mediaMat[$materia] = $s / $numMat[$materia];
    if ($mediaMat[$materia] < $min) $min = $mediaMat[$materia];
}
echo "<h2>6) Materia con la media piu' bassa</h2>";
foreach ($mediaMat as $materia => $m)
    if ($m == $min) printf("%s (%.2f)<br>", $materia, $m);
?>
