<?php
// =========================================================
//  VERIFICA 2 - NEGOZIO     (soluzione con PHP base)
// =========================================================

$righe = file("../vendite.txt");

for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $data      = $parti[0];
    $prodotto  = $parti[1];
    $categoria = $parti[2];
    $quantita  = $parti[3];
    $prezzo    = $parti[4];
    settype($quantita, "integer");
    settype($prezzo, "double");          // ha la virgola (il punto): e' un decimale!

    $incasso = $quantita * $prezzo;

    // salvo la riga per il punto 4
    $vendite[] = array($data, $prodotto, $quantita, $incasso);

    // 1) incasso per prodotto (Schema C: sommare)
    if (!isset($incassoProd[$prodotto])) $incassoProd[$prodotto] = 0;
    $incassoProd[$prodotto] = $incassoProd[$prodotto] + $incasso;

    // 2) incasso per categoria
    if (!isset($incassoCat[$categoria])) $incassoCat[$categoria] = 0;
    $incassoCat[$categoria] = $incassoCat[$categoria] + $incasso;

    // 3) pezzi venduti per prodotto
    if (!isset($pezzi[$prodotto])) $pezzi[$prodotto] = 0;
    $pezzi[$prodotto] = $pezzi[$prodotto] + $quantita;

    // 5) numero di vendite per categoria (Schema B: contare)
    if (!isset($numVendite[$categoria])) $numVendite[$categoria] = 0;
    $numVendite[$categoria] = $numVendite[$categoria] + 1;

    // 6) incasso per giorno
    if (!isset($incassoGiorno[$data])) $incassoGiorno[$data] = 0;
    $incassoGiorno[$data] = $incassoGiorno[$data] + $incasso;
}

// ---------- 1) CLASSIFICA INCASSI (Schema F) ----------
arsort($incassoProd);
echo "<h2>1) Incasso per prodotto</h2>";
echo "<table border=1><tr><th>Pos</th><th>Prodotto</th><th>Incasso</th></tr>";
$pos = 1;
foreach ($incassoProd as $prodotto => $euro)
{
    printf("<tr><td>%d</td><td>%s</td><td>%.2f &euro;</td></tr>", $pos, $prodotto, $euro);
    $pos++;
}
echo "</table>";

// ---------- 2) CATEGORIA CON INCASSO MAGGIORE (Schema E) ----------
$max = -1;
foreach ($incassoCat as $categoria => $euro)
    if ($euro > $max) $max = $euro;
echo "<h2>2) Categoria con l'incasso maggiore</h2>";
foreach ($incassoCat as $categoria => $euro)
    if ($euro == $max) printf("%s (%.2f &euro;)<br>", $categoria, $euro);

// ---------- 3) PRODOTTO PIU' VENDUTO (Schema E) ----------
$max = -1;
foreach ($pezzi as $prodotto => $n)
    if ($n > $max) $max = $n;
echo "<h2>3) Prodotto piu' venduto</h2>";
foreach ($pezzi as $prodotto => $n)
    if ($n == $max) echo "$prodotto ($n pezzi)<br>";

// ---------- 4) FORM CON CASELLA DI TESTO ----------
echo "<h2>4) Cerca un prodotto</h2>";
echo "<form action='' method='get'>";
echo "Prodotto: <input type='text' name='prodotto'> <input type='submit' value='Cerca'>";
echo "</form>";

if (isset($_GET["prodotto"]))
{
    $cerca = trim($_GET["prodotto"]);     // tolgo eventuali spazi scritti per sbaglio
    $trovato = false;
    $totale = 0;
    echo "<table border=1><tr><th>Data</th><th>Quantita</th><th>Incasso</th></tr>";
    for ($i = 0; $i < count($vendite); $i++)
    {
        if ($vendite[$i][1] == $cerca)
        {
            printf("<tr><td>%s</td><td>%d</td><td>%.2f</td></tr>", $vendite[$i][0], $vendite[$i][2], $vendite[$i][3]);
            $totale = $totale + $vendite[$i][3];
            $trovato = true;
        }
    }
    echo "</table>";
    if ($trovato)
        printf("Totale %s: %.2f &euro;", $cerca, $totale);
    else
        echo "Prodotto non trovato";
}

// ---------- 5) FILE categorie.html (Schema G + J) ----------
ksort($numVendite);
$html = "<html><body><h1>Categorie</h1><ul>\n";
foreach ($numVendite as $categoria => $n)
    $html = $html . "<li>$categoria: $n vendite</li>\n";
$html = $html . "</ul></body></html>";
file_put_contents("categorie.html", $html);
echo "<h2>5) File creato</h2><a href='categorie.html'>Apri categorie.html</a>";

// ---------- 6) FACOLTATIVO: INCASSO PER GIORNO ----------
echo "<h2>6) Incasso per giorno</h2>";
$max = -1;
foreach ($incassoGiorno as $data => $euro)
{
    printf("%s: %.2f &euro;<br>", $data, $euro);
    if ($euro > $max) $max = $euro;
}
foreach ($incassoGiorno as $data => $euro)
    if ($euro == $max) printf("<b>Giorno migliore: %s (%.2f &euro;)</b><br>", $data, $euro);
?>
