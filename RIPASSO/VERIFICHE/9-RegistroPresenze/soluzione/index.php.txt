<?php
// =========================================================
//  VERIFICA 9 - REGISTRO PRESENZE     (soluzione con PHP base)
//  Novita': l'ultimo campo e' TESTO -> serve trim()
// =========================================================

$righe = file("../presenze.txt");

for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $data     = $parti[0];
    $studente = $parti[1];
    $stato    = trim($parti[2]);     // "A\n" -> "A"  (settype qui non va: non e' un numero!)

    // ARRAY A DUE LIVELLI: $registro["Rossi"]["03/11"] = "P"  (per il punto 5)
    $registro[$studente][$data] = $stato;

    // elenco delle date (chiave = data, cosi' non si ripetono)
    $date[$data] = 1;

    // 1) contatori per studente
    if (!isset($assenze[$studente]))
    {
        $assenze[$studente] = 0;
        $ritardi[$studente] = 0;
        $giorni[$studente]  = 0;
    }
    $giorni[$studente] = $giorni[$studente] + 1;
    if ($stato == "A") $assenze[$studente] = $assenze[$studente] + 1;
    if ($stato == "R") $ritardi[$studente] = $ritardi[$studente] + 1;

    // 3) assenti per giorno
    if (!isset($assentiGiorno[$data])) $assentiGiorno[$data] = 0;
    if ($stato == "A") $assentiGiorno[$data] = $assentiGiorno[$data] + 1;
}

// ---------- 1) e 2) TABELLA ASSENZE ----------
echo "<h2>1) e 2) Assenze e ritardi</h2>";
echo "<table border=1><tr><th>Studente</th><th>Assenze</th><th>Ritardi</th><th>% assenze</th></tr>";
foreach ($assenze as $studente => $a)
{
    $perc = $a / $giorni[$studente] * 100;
    if ($perc > 25)                                        // 2) a rischio
        printf("<tr><td><font color=red>%s</font></td><td>%d</td><td>%d</td><td><font color=red>%.1f%%</font></td></tr>",
               $studente, $a, $ritardi[$studente], $perc);
    else
        printf("<tr><td>%s</td><td>%d</td><td>%d</td><td>%.1f%%</td></tr>",
               $studente, $a, $ritardi[$studente], $perc);
}
echo "</table>";

// ---------- 3) GIORNO CON PIU' ASSENTI ----------
$max = -1;
foreach ($assentiGiorno as $data => $n)
    if ($n > $max) $max = $n;
echo "<h2>3) Giorno con piu' assenti</h2>";
foreach ($assentiGiorno as $data => $n)
    if ($n == $max) echo "$data ($n assenti)<br>";

// ---------- 4) RICERCA SENZA MAIUSCOLE/MINUSCOLE ----------
echo "<h2>4) Cerca uno studente</h2><form action='' method='get'>";
echo "Cognome: <input type='text' name='cognome'> <input type='submit' value='Cerca'></form>";

if (isset($_GET["cognome"]))
{
    $cerca = strtoupper(trim($_GET["cognome"]));     // "rossi " -> "ROSSI"
    $trovato = false;
    foreach ($registro as $studente => $giorniStud)
    {
        if (strtoupper($studente) == $cerca)          // confronto tutto in MAIUSCOLO
        {
            $trovato = true;
            echo "<b>$studente</b><br>";
            foreach ($giorniStud as $data => $stato)
            {
                if ($stato == "A") echo "$data: assente<br>";
                if ($stato == "R") echo "$data: ritardo<br>";
            }
        }
    }
    if (!$trovato) echo "Studente non trovato";
}

// ---------- 5) FILE registro.html (tabella a due dimensioni) ----------
ksort($registro);
$html = "<html><body><h1>Registro</h1><table border=1>\n<tr><th>Studente</th>";
foreach ($date as $data => $x)
    $html = $html . "<th>$data</th>";               // intestazione: le date
$html = $html . "</tr>\n";

foreach ($registro as $studente => $giorniStud)
{
    $html = $html . "<tr><td>$studente</td>";
    foreach ($date as $data => $x)
    {
        $stato = $giorniStud[$data];
        if ($stato == "A")     $html = $html . "<td><font color=red>A</font></td>";
        elseif ($stato == "R") $html = $html . "<td><font color=orange>R</font></td>";
        else                   $html = $html . "<td>P</td>";
    }
    $html = $html . "</tr>\n";
}
$html = $html . "</table></body></html>";
file_put_contents("registro.html", $html);
echo "<h2>5) File creato</h2><a href='registro.html'>Apri registro.html</a>";

// ---------- 6) FACOLTATIVO: SEMPRE PRESENTI ----------
echo "<h2>6) Sempre presenti</h2>";
foreach ($assenze as $studente => $a)
    if ($a == 0 && $ritardi[$studente] == 0) echo "$studente<br>";
?>
