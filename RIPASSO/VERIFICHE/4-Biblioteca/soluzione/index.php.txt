<?php
// =========================================================
//  VERIFICA 4 - BIBLIOTECA     (soluzione con PHP base)
// =========================================================

$righe = file("../prestiti.txt");

for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $titolo = $parti[0];
    $autore = $parti[1];
    $genere = $parti[2];
    $utente = $parti[3];
    $giorni = $parti[4];
    settype($giorni, "integer");

    $prestiti[] = array($titolo, $autore, $genere, $utente, $giorni);

    // 1) quante volte ogni libro (contare)
    if (!isset($volte[$titolo])) $volte[$titolo] = 0;
    $volte[$titolo] = $volte[$titolo] + 1;

    // 2) quanti libri ogni utente (contare) + 6) giorni totali per utente
    if (!isset($libriUtente[$utente]))
    {
        $libriUtente[$utente] = 0;
        $giorniUtente[$utente] = 0;
    }
    $libriUtente[$utente]  = $libriUtente[$utente] + 1;
    $giorniUtente[$utente] = $giorniUtente[$utente] + $giorni;

    // 4) generi (mi serve solo la lista: uso il genere come chiave)
    $generi[$genere] = 1;

    // 5) giorni per autore (sommare)
    if (!isset($giorniAutore[$autore])) $giorniAutore[$autore] = 0;
    $giorniAutore[$autore] = $giorniAutore[$autore] + $giorni;
}

// ---------- 1) LIBRI PIU' PRESTATI (Schema F) ----------
arsort($volte);
echo "<h2>1) Libri piu' prestati</h2>";
echo "<table border=1><tr><th>Titolo</th><th>Prestiti</th></tr>";
foreach ($volte as $titolo => $n)
    echo "<tr><td>$titolo</td><td>$n</td></tr>";
echo "</table>";

// ---------- 2) UTENTE CON PIU' PRESTITI (Schema E) ----------
$max = -1;
foreach ($libriUtente as $utente => $n)
    if ($n > $max) $max = $n;
echo "<h2>2) Utente con piu' prestiti</h2>";
foreach ($libriUtente as $utente => $n)
    if ($n == $max) echo "$utente ($n libri)<br>";

// ---------- 3) PRESTITI IN RITARDO ----------
echo "<h2>3) Prestiti in ritardo (piu' di 30 giorni)</h2>";
$ritardi = 0;
for ($i = 0; $i < count($prestiti); $i++)
{
    if ($prestiti[$i][4] > 30)
    {
        echo $prestiti[$i][0]." - ".$prestiti[$i][3]." - ".$prestiti[$i][4]." giorni<br>";
        $ritardi++;
    }
}
echo "<b>Totale ritardi: $ritardi</b>";

// ---------- 4) FORM CON I GENERI ----------
echo "<h2>4) Libri per genere</h2>";
echo "<form action='' method='get'><select name='genere'>";
foreach ($generi as $genere => $x)
    echo "<option value='$genere'>$genere</option>";
echo "</select> <input type='submit' value='Mostra'></form>";

if (isset($_GET["genere"]))
{
    $scelto = $_GET["genere"];
    $giaScritti = array();                  // titoli gia' stampati
    echo "<ul>";
    for ($i = 0; $i < count($prestiti); $i++)
    {
        if ($prestiti[$i][2] == $scelto && !in_array($prestiti[$i][0], $giaScritti))
        {
            echo "<li>".$prestiti[$i][0]."</li>";
            $giaScritti[] = $prestiti[$i][0];   // me lo segno per non ripeterlo
        }
    }
    echo "</ul>";
}

// ---------- 5) FILE autori.html (Schema G + J) ----------
ksort($giorniAutore);
$html = "<html><body><h1>Autori</h1><table border=1>\n";
foreach ($giorniAutore as $autore => $g)
    $html = $html . "<tr><td>$autore</td><td>$g giorni</td></tr>\n";
$html = $html . "</table></body></html>";
file_put_contents("autori.html", $html);
echo "<h2>5) File creato</h2><a href='autori.html'>Apri autori.html</a>";

// ---------- 6) FACOLTATIVO: MEDIA GIORNI PER UTENTE ----------
echo "<h2>6) Durata media dei prestiti per utente</h2>";
foreach ($giorniUtente as $utente => $g)
{
    $media = $g / $libriUtente[$utente];
    if ($media > 30)
        printf("<font color=red>%s: %.1f giorni</font><br>", $utente, $media);
    else
        printf("%s: %.1f giorni<br>", $utente, $media);
}
?>
