<?php
// =========================================================
//  VERIFICA 10 - ELEZIONI     (soluzione con PHP base)
//  Novita': il programma SCRIVE nel file dei dati
// =========================================================

// ---------- 4) PRIMA DI TUTTO registro un eventuale nuovo voto ----------
// (lo faccio all'inizio, cosi' i risultati qui sotto sono gia' aggiornati)
if (isset($_GET["classe"]) && isset($_GET["candidato"]))
{
    $righe = file("../schede.txt");
    $testo = implode("", $righe);          // rimetto insieme il file (le righe hanno gia' il \n)
    $testo = $testo . $_GET["classe"] . "," . $_GET["candidato"] . "\n";   // aggiungo la nuova riga
    file_put_contents("../schede.txt", $testo);                            // riscrivo il file
    echo "<p><b><font color=green>Voto registrato</font></b></p>";
}

// ---------- LETTURA ----------
$righe = file("../schede.txt");
$totale = count($righe);

for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $classe    = $parti[0];
    $candidato = trim($parti[1]);          // ultimo campo di testo: trim!

    // 1) voti per candidato
    if (!isset($voti[$candidato])) $voti[$candidato] = 0;
    $voti[$candidato] = $voti[$candidato] + 1;

    // 2) ARRAY A DUE LIVELLI: $perClasse["5A"]["Rossi"] = voti
    if (!isset($perClasse[$classe][$candidato])) $perClasse[$classe][$candidato] = 0;
    $perClasse[$classe][$candidato] = $perClasse[$classe][$candidato] + 1;

    // 6) schede per classe
    if (!isset($schede[$classe])) $schede[$classe] = 0;
    $schede[$classe] = $schede[$classe] + 1;
}

// ---------- 1) RISULTATI CON PERCENTUALE ----------
arsort($voti);
echo "<h2>1) Risultati ($totale schede)</h2><table border=1>";
foreach ($voti as $candidato => $n)
    printf("<tr><td>%s</td><td>%d</td><td>%.1f%%</td></tr>", $candidato, $n, $n / $totale * 100);
echo "</table>";

// ---------- 2) PIU' VOTATO IN OGNI CLASSE ----------
echo "<h2>2) Piu' votato per classe</h2>";
foreach ($perClasse as $classe => $candidati)
{
    $max = -1;
    foreach ($candidati as $candidato => $n)
        if ($n > $max) $max = $n;
    echo "$classe: ";
    foreach ($candidati as $candidato => $n)
        if ($n == $max) echo "$candidato ($n) ";
    echo "<br>";
}

// ---------- 3) I DUE ELETTI ----------
echo "<h2>3) Eletti</h2>";
$pos = 1;
foreach ($voti as $candidato => $n)     // $voti e' gia' ordinato (arsort)
{
    if ($pos <= 2) echo "$pos&deg; eletto: $candidato<br>";
    $pos++;
}

// ---------- 4) FORM PER VOTARE ----------
echo "<h2>4) Vota</h2><form action='' method='get'>";
echo "Classe: <select name='classe'>";
foreach ($schede as $classe => $n)
    echo "<option value='$classe'>$classe</option>";
echo "</select><br>";
ksort($voti);                            // candidati in ordine alfabetico nel form
foreach ($voti as $candidato => $n)
    echo "<input type='radio' name='candidato' value='$candidato' required> $candidato<br>";
echo "<input type='submit' value='Vota'></form>";

// ---------- 5) FILE risultati.html ----------
// ($voti e' gia' in ordine alfabetico grazie al ksort qui sopra)
$html = "<html><body><h1>Risultati elezioni</h1><table border=1>\n";
foreach ($voti as $candidato => $n)
    $html = $html . "<tr><td>$candidato</td><td>$n</td></tr>\n";
$html = $html . "</table></body></html>";
file_put_contents("risultati.html", $html);
echo "<h2>5) File creato</h2><a href='risultati.html'>Apri risultati.html</a>";

// ---------- 6) FACOLTATIVO ----------
$max = -1;
foreach ($schede as $classe => $n)
    if ($n > $max) $max = $n;
echo "<h2>6) Classe che ha votato di piu'</h2>";
foreach ($schede as $classe => $n)
    if ($n == $max) echo "$classe ($n schede)<br>";
?>
