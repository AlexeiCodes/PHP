<?php
// =========================================================
//  VOTI DELLA CLASSE - soluzione con PHP base
// =========================================================

$righe = file("../voti.txt");
$sommaTutti = 0;

// ---------- LETTURA ----------
for ($i = 0; $i < count($righe); $i++)
{
    $parti = explode(",", $righe[$i]);
    $stud    = $parti[0];
    $materia = $parti[1];
    $voto    = $parti[2];
    settype($voto, "integer");

    $voti[] = array($stud, $materia, $voto);
    $sommaTutti = $sommaTutti + $voto;

    // caselle per studente
    if (!isset($somma[$stud]))
    {
        $somma[$stud] = 0;  $num[$stud] = 0;  $insuffStud[$stud] = 0;
        $studenti[] = $stud;
    }
    $somma[$stud] = $somma[$stud] + $voto;
    $num[$stud]   = $num[$stud] + 1;

    // caselle per materia
    if (!isset($insuffMat[$materia]))
    {
        $insuffMat[$materia] = 0;
        $materie[] = $materia;
    }

    if ($voto < 6)
    {
        $insuffMat[$materia]++;
        $insuffStud[$stud]++;
    }
}

// ---------- 1) MEDIA DI OGNI STUDENTE ----------
echo "<h2>1) Medie</h2>";
for ($i = 0; $i < count($studenti); $i++)
{
    $s = $studenti[$i];
    $media[$s] = $somma[$s] / $num[$s];      // la salvo: mi serve al punto 3
    printf("%s: %.2f<br>", $s, $media[$s]);
}

// ---------- 2) MEDIA DELLA CLASSE ----------
$mediaClasse = $sommaTutti / count($righe);
printf("<h2>2) Media della classe: %.2f</h2>", $mediaClasse);

// ---------- 3) SOPRA LA MEDIA DELLA CLASSE ----------
// non si puo' fare durante la lettura: la media della classe la conosco solo ALLA FINE
echo "<h2>3) Sopra la media della classe</h2>";
for ($i = 0; $i < count($studenti); $i++)
    if ($media[$studenti[$i]] > $mediaClasse)
        printf("%s (%.2f)<br>", $studenti[$i], $media[$studenti[$i]]);

// ---------- 4) INSUFFICIENZE PER MATERIA + MASSIMO ----------
echo "<h2>4) Insufficienze per materia</h2>";
$max = -1;
for ($i = 0; $i < count($materie); $i++)
{
    $m = $materie[$i];
    echo "$m: $insuffMat[$m]<br>";
    if ($insuffMat[$m] > $max) $max = $insuffMat[$m];
}
echo "Materia/e con piu' insufficienze: ";
for ($i = 0; $i < count($materie); $i++)
    if ($insuffMat[$materie[$i]] == $max) echo $materie[$i] . " ";

// ---------- 5) ORDINE ALFABETICO A MANO ----------
for ($i = 0; $i < count($studenti) - 1; $i++)
    for ($j = 0; $j < count($studenti) - 1 - $i; $j++)
        if ($studenti[$j] > $studenti[$j + 1])        // le stringhe si confrontano in ordine alfabetico
        {
            $tmp = $studenti[$j];
            $studenti[$j] = $studenti[$j + 1];
            $studenti[$j + 1] = $tmp;
        }
echo "<h2>5) Esito</h2>";
for ($i = 0; $i < count($studenti); $i++)
{
    $s = $studenti[$i];
    if ($insuffStud[$s] == 0) echo "$s: PROMOSSO<br>";
    else echo "$s: DEBITI: $insuffStud[$s]<br>";
}

// ---------- 6) RICERCA SENZA MAIUSCOLE ----------
echo "<h2>6) Cerca uno studente</h2><form action='' method='get'>";
echo "Cognome: <input type='text' name='cognome'> <input type='submit' value='Cerca'></form>";
if (isset($_GET["cognome"]))
{
    $cerca = strtoupper(trim($_GET["cognome"]));
    $trovato = 0;
    for ($i = 0; $i < count($voti); $i++)
        if (strtoupper($voti[$i][0]) == $cerca)
        {
            echo $voti[$i][1] . ": " . $voti[$i][2] . "<br>";
            $trovato = 1;
        }
    if ($trovato == 0) echo "Studente non trovato";
}
?>
